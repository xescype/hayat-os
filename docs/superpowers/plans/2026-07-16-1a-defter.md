**Goal:** Sistemin omurgası Defter'i (append-only ledger) + ekranını kur; kullanıcı metin-notu ve sistem olayları (login, yedek) yazsın, aranıp filtrelensin, dışa aktarılsın.

## Global Constraints

- Test/build **Docker/Linux konteynerinde** koşar (Windows host esbuild bloklu). Lokal: `cd infra && docker compose up -d db` sonra konteynerde test. CI: Postgres service'li job.
- Konteynerde **pnpm 9** (packageManager pnpm@9.15.4). Migration **elle** yazılır (`NNNN_*.sql` + `meta/_journal.json`), drizzle-kit yok.
- **Ledger append-only:** UPDATE/DELETE yasak; yalnız `superseded_by` NULL→id (tek sefer). DB trigger zorlar.
- **supersede atomik:** `UPDATE ... WHERE superseded_by IS NULL` + rowcount=0 → tx rollback.
- **Türkçe-I:** arama `normalizeSearch` (ASCII fold, ç→c ş→s ğ→g ü→u ö→o ı/İ→i, lowercase) → `search_norm`; yazım+sorgu aynı fonksiyon.

- **Ürün adı "Oda"** kullanıcı-yüzünde (manifest, başlık, bildirim); kod adı `hayat-os`/`@hayatos/*`/imaj DEĞİŞMEZ.
- Ledger route'ları oturum korumalı (`readSession`).
- Zaman: `ts`=olay anı (override edilebilir), `created_at`=DB anı (değişmez). Düz-metin export Europe/Istanbul; JSON ISO-8601 UTC.

// packages/shared/src/normalize.ts
// Turkce karakter -> ASCII. Map ONCE, sonra lowercase (JS 'İ'.toLowerCase() combining-dot uretir).
const FOLD: Record<string, string> = {
  ç: "c", Ç: "c", ş: "s", Ş: "s", ğ: "g", Ğ: "g",
  ü: "u", Ü: "u", ö: "o", Ö: "o", ı: "i", İ: "i",
  â: "a", Â: "a", î: "i", Î: "i", û: "u", Û: "u",
};

/** Arama normalizasyonu: Turkce->ASCII katla, lowercase. Yazim+sorgu ayni fonksiyon. */
export function normalizeSearch(input: string): string {
  let out = "";
  for (const ch of input) out += FOLD[ch] ?? ch;
  return out.toLowerCase();
}
```

// apps/api/src/modules/ledger/service.ts
import { ledgerWriteSchema, normalizeSearch, type LedgerWriteInput } from "@hayatos/shared";
import { db } from "../../db/client.js";
import { ledgerEntries } from "../../db/schema.js";

function buildRow(input: LedgerWriteInput) {
  const parsed = ledgerWriteSchema.parse(input);
  const searchNorm = normalizeSearch(`${parsed.raw_input ?? ""} ${parsed.response_text ?? ""}`.trim());
  return {
    channel: parsed.channel,
    actor: parsed.actor,
    rawInput: parsed.raw_input ?? null,
    interpretation: (parsed.interpretation ?? null) as unknown,
    moduleWrites: parsed.module_writes ?? null,
    responseText: parsed.response_text ?? null,
    audioRef: parsed.audio_ref ?? null,
    sensitivity: parsed.sensitivity,
    searchNorm: searchNorm.length ? searchNorm : null,
    ...(parsed.ts ? { ts: parsed.ts } : {}),
  };
}

/** Defter'e append-only yazim. Tek yazma kapisi (Bolum 5). */
export async function writeLedger(input: LedgerWriteInput): Promise<{ id: string }> {
  const [row] = await db.insert(ledgerEntries).values(buildRow(input)).returning({ id: ledgerEntries.id });
  return { id: row!.id };
}
```

import { and, eq, isNull } from "drizzle-orm";

export type LedgerRow = typeof ledgerEntries.$inferSelect;

/** Atomik supersede: yeni satir + eskiyi isaretle (WHERE superseded_by IS NULL). rowcount=0 => rollback. */
export async function supersede(oldId: string, newInput: LedgerWriteInput): Promise<{ id: string }> {
  return db.transaction(async (tx) => {
    const [created] = await tx.insert(ledgerEntries).values(buildRow(newInput)).returning({ id: ledgerEntries.id });
    const marked = await tx
      .update(ledgerEntries)
      .set({ supersededBy: created!.id })
      .where(and(eq(ledgerEntries.id, oldId), isNull(ledgerEntries.supersededBy)))
      .returning({ id: ledgerEntries.id });
    if (marked.length === 0) throw new Error("Kayit bulunamadi veya zaten duzeltilmis.");
    return { id: created!.id };
  });
}

/** Supersede zincirini (en eski -> en yeni) dondurur. */
export async function getChain(startId: string): Promise<LedgerRow[]> {
  const out: LedgerRow[] = [];
  let cur: string | null = startId;
  const seen = new Set<string>();
  while (cur && !seen.has(cur)) {
    seen.add(cur);
    const [row] = await db.select().from(ledgerEntries).where(eq(ledgerEntries.id, cur));
    if (!row) break;
    out.push(row);
    cur = row.supersededBy;
  }
  return out;
}

// apps/api/src/modules/ledger/cursor.ts
// Cursor = base64url("<ISO ts>|<id>"). ts desc + id tie-break sayfalama.
export function encodeCursor(ts: Date, id: string): string {
  return Buffer.from(`${ts.toISOString()}|${id}`, "utf8").toString("base64url");
}
export function decodeCursor(c: string): { ts: Date; id: string } | null {
  try {
    const [iso, id] = Buffer.from(c, "base64url").toString("utf8").split("|");
    if (!iso || !id) return null;
    const ts = new Date(iso);
    if (Number.isNaN(ts.getTime())) return null;
    return { ts, id };
  } catch {
    return null;
  }
}

import { desc, gte, lte, ilike, sql as dsql } from "drizzle-orm";
import { normalizeSearch } from "@hayatos/shared";
import { encodeCursor, decodeCursor } from "./cursor.js";

export interface ListOpts { cursor?: string; limit?: number; }
export interface SearchOpts extends ListOpts { q?: string; from?: Date; to?: Date; channel?: string; }
export interface Page { rows: LedgerRow[]; nextCursor: string | null; }

const clampLimit = (n?: number) => Math.min(Math.max(n ?? 50, 1), 200);

export async function listActive(opts: ListOpts = {}): Promise<Page> {
  return search({ ...opts });
}

export async function search(opts: SearchOpts = {}): Promise<Page> {
  const limit = clampLimit(opts.limit);
  const conds = [isNull(ledgerEntries.supersededBy)];
  if (opts.q && opts.q.trim()) conds.push(ilike(ledgerEntries.searchNorm, `%${normalizeSearch(opts.q)}%`));
  if (opts.from) conds.push(gte(ledgerEntries.ts, opts.from));
  if (opts.to) conds.push(lte(ledgerEntries.ts, opts.to));
  if (opts.channel) conds.push(eq(ledgerEntries.channel, opts.channel));
  const cur = opts.cursor ? decodeCursor(opts.cursor) : null;
  if (cur) conds.push(dsql`(${ledgerEntries.ts}, ${ledgerEntries.id}) < (${cur.ts}, ${cur.id})`);

  const rows = await db
    .select()
    .from(ledgerEntries)
    .where(and(...conds))
    .orderBy(desc(ledgerEntries.ts), desc(ledgerEntries.id))
    .limit(limit + 1);

  const hasMore = rows.length > limit;
  const page = hasMore ? rows.slice(0, limit) : rows;
  const last = page[page.length - 1];
  return { rows: page, nextCursor: hasMore && last ? encodeCursor(last.ts, last.id) : null };
}

// apps/api/src/modules/ledger/format.ts
import type { LedgerRow } from "./service.js";

const TZ = "Europe/Istanbul";
const dt = new Intl.DateTimeFormat("tr-TR", {
  timeZone: TZ, day: "2-digit", month: "2-digit", hour: "2-digit", minute: "2-digit",
});
const dOnly = new Intl.DateTimeFormat("tr-TR", { timeZone: TZ, day: "2-digit", month: "2-digit", year: "numeric" });

function line(r: Pick<LedgerRow, "ts" | "rawInput" | "responseText">): string {
  const parts = dt.formatToParts(r.ts);
  const g = (t: string) => parts.find((p) => p.type === t)?.value ?? "";
  const stamp = `${g("day")}.${g("month")} ${g("hour")}:${g("minute")}`;
  const segs: string[] = [];
  if (r.rawInput) segs.push(`Sen: ${r.rawInput}`);
  if (r.responseText) segs.push(`Sistem: ${r.responseText}`);
  return `${stamp} — ${segs.join(" / ")}`;
}

/** Insan okunur cizelge: baslik + aktif satirlar (Istanbul saati). */
export function toPlainText(rows: Pick<LedgerRow, "ts" | "rawInput" | "responseText">[], from?: Date, to?: Date): string {
  const header =
    from && to ? `Defter — ${dOnly.format(from)} → ${dOnly.format(to)}, Europe/Istanbul` : `Defter — tümü, Europe/Istanbul`;
  return [header, ...rows.map(line)].join("\n");
}

import { toPlainText } from "./format.js";

export interface ExportOpts { from?: Date; to?: Date; format: "json" | "text"; }
export async function exportRange(opts: ExportOpts): Promise<{ body: string; contentType: string; filename: string }> {
  const conds = [];
  if (opts.from) conds.push(gte(ledgerEntries.ts, opts.from));
  if (opts.to) conds.push(lte(ledgerEntries.ts, opts.to));
  const tag = `${opts.from?.toISOString().slice(0, 10) ?? "all"}_${opts.to?.toISOString().slice(0, 10) ?? "all"}`;

  if (opts.format === "json") {
    // TAM kayit: aktif + superseded, ts artan.
    const rows = await db.select().from(ledgerEntries).where(conds.length ? and(...conds) : undefined).orderBy(ledgerEntries.ts);
    return { body: JSON.stringify(rows, null, 2), contentType: "application/json", filename: `defter-${tag}.json` };
  }
  // duz metin: yalniz aktif.
  const rows = await db
    .select()
    .from(ledgerEntries)
    .where(and(isNull(ledgerEntries.supersededBy), ...conds))
    .orderBy(ledgerEntries.ts);
  return { body: toPlainText(rows, opts.from, opts.to), contentType: "text/plain", filename: `defter-${tag}.txt` };
}

```tsx
// apps/pwa/src/views/Defter.tsx
import { useEffect, useState } from "react";
import { Card, Button, Input, LedgerRow } from "../components/ui.js";
import { ledgerApi, type LedgerRow as Row } from "../lib/ledger.js";

export default function Defter() {
  const [rows, setRows] = useState<Row[]>([]);
  const [q, setQ] = useState("");
  const [note, setNote] = useState("");
  const [loaded, setLoaded] = useState(false);

  async function load(query = "") {
    const page = await ledgerApi.list({ q: query || undefined });
    setRows(page.rows);
    setLoaded(true);
  }
  useEffect(() => { load(); }, []);

  async function addNote() {
    const t = note.trim();
    if (!t) return;
    await ledgerApi.note(t);
    setNote("");
    await load(q);
  }

  return (
    <div className="w-full max-w-md space-y-4">
      <Card className="space-y-3">
        <div className="flex gap-2">
          <Input placeholder="Not yaz…" value={note} onChange={(e) => setNote(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && addNote()} />
          <Button onClick={addNote} disabled={!note.trim()}>Ekle</Button>
        </div>
        <Input placeholder="Ara…" value={q}
          onChange={(e) => { setQ(e.target.value); load(e.target.value); }} />
      </Card>
      <Card>
        {!loaded ? (
          <div className="text-sm text-text-dim">Yükleniyor…</div>
        ) : rows.length === 0 ? (
          <div className="text-sm text-text-faint">{q ? "Bu filtreyle kayıt yok." : "Defter boş — ilk notu yaz."}</div>
        ) : (
          rows.map((r) => (
            <LedgerRow key={r.id} ts={r.ts} channel={r.channel} sen={r.rawInput} sistem={r.responseText}
              sensitive={r.sensitivity === "sensitive"} />
          ))
        )}
      </Card>
    </div>
  );
}
```
